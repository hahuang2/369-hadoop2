Part 1(CountryJoin):

rm -rf countryjoin/join_output countryjoin/sum_output countryjoin/sort_output
mkdir -p countryjoin
./gradlew run --args="HostCountryJoin hostname_country input_access_log countryjoin/join_output"
./gradlew run --args="Sum countryjoin/join_output countryjoin/sum_output"
./gradlew run --args="SortDes countryjoin/sum_output countryjoin/sort_output"
